3. EVALUATION OPTIONS
---------------------

# Do not evaluate:
--evaluate false

# Evaluate against a different folder bot:
--evaluate true \
--evaluation-opponent-folder "$PWD/path/to/another-agent"

# Override the opponent's deck.csv without modifying its source folder:
--evaluation-opponent-deck "$PWD/path/to/opponent-deck.csv"

# Use another deck for the trained model:
--evaluation-own-deck "$PWD/path/to/our-deck.csv"

# Use the latest checkpoint instead of the best-validation checkpoint:
--evaluation-checkpoint latest

# If evaluation is disabled, the combined config JSON contains exactly:
# "NO EVALUATION PERFORMED FOR THIS RUN"

10. ALPHAZERO SELF-PLAY GENERATION AND LATEST-NETWORK UPDATE
-------------------------------------------------------------

# Start from a completed imitation checkpoint that has:
#   value_activation=tanh
# Package/evaluation search can be enabled explicitly at build time, but the
# self-play trainer requires search-enabled metadata. To create that initial
# checkpoint, run the IL command with --tree-search true and these switches.
#
# AlphaZero initial profile: 800 simulations, root Dirichlet noise during
# self-play, visit sampling at temperature 1.0, no hard top-k pruning.

REPLAY_ROOT="$PWD/Imitation-Learning/engine-native-selfplay/replay-v1"
LATEST="$PWD/Imitation-Learning/engine-native-training/il-search-seed/checkpoint.best.pt"
DECK="$PWD/Rising-Tide-Fixed-Metal-v15/deck.csv"

GEN_JOB=$(sbatch --parsable \
  Engine-Native-Architecture/cluster/generate_selfplay.sbatch \
  --checkpoint "$LATEST" \
  --deck "$DECK" \
  --output-dir "$REPLAY_ROOT" \
  --run-name selfplay-v1 \
  --model-name "Engine Native AlphaZero v1" \
  --model-description "Latest-network self-play; no promotion gate" \
  --games 100 \
  --mcts-simulations 800 \
  --mcts-max-depth 32 \
  --mcts-c-puct 1.5 \
  --mcts-dirichlet-alpha 0.3 \
  --mcts-dirichlet-epsilon 0.25 \
  --mcts-temperature 1.0 \
  --mcts-seed 20260730)
GEN_JOB=${GEN_JOB%%;*}

UPDATE_DIR="$PWD/Imitation-Learning/engine-native-selfplay/update-v1"
TRAIN_JOB=$(sbatch --parsable \
  --dependency="afterok:$GEN_JOB" \
  Engine-Native-Architecture/cluster/train_selfplay.sbatch \
  --replay-root "$REPLAY_ROOT" \
  --out-dir "$UPDATE_DIR" \
  --initial-checkpoint "$LATEST" \
  --replay-window-games 500000 \
  --validation-fraction 0.10 \
  --epochs 1 \
  --batch-size 256 \
  --learning-rate 1e-3 \
  --l2-weight 1e-4 \
  --seed 20260730)
TRAIN_JOB=${TRAIN_JOB%%;*}

echo "Generate: $GEN_JOB"
echo "Update: $TRAIN_JOB (afterok:$GEN_JOB)"

# For the next cycle, point --checkpoint/--initial-checkpoint at:
# "$UPDATE_DIR/checkpoint.latest.pt"
# Each generated game pins one checkpoint hash from start to finish. The update
# always becomes the next latest network; there is no 55% promotion gate.
# Use --mcts-per-decision-seconds and --mcts-game-budget-seconds for bounded
# generation. Kaggle packaging should normally reserve headroom within the
# ten-minute game limit, for example 20 seconds per decision and 480 total.
